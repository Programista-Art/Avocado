// PĘTLA: Pobieranie wiadomości z serwera co 2 sekundy (GET)
wykonuj_co(2000, funkcja()
poczatek
    pobierz_z_sieci("http://localhost:8080/api/czat/czytaj", funkcja(surowy_tekst)
    poczatek
        zmienna okno = wyszukaj_element("#ekran-czatu");
        
        // 1. Czyścimy stare wiadomości (żeby nie dodawać ich w nieskończoność do starych)
        ustaw_tekst(okno, ""); 
        
        // 2. Ożywiamy JSON - zamieniamy tekst na prawdziwą listę!
        zmienna lista_wiadomosci = parsuj_json(surowy_tekst);
        
        // 3. Pobieramy długość listy (w JS to pole nazywa się "length")
        zmienna ilosc = pobierz_pole(lista_wiadomosci, "length");
        
        // 4. Pętla przechodząca przez wszystkie wiadomości
        powtarzaj(ilosc)
        poczatek
            // Wyciągamy konkretny obiekt wiadomości używając indeksu 'i' z pętli

            // Wyciągamy autora i tekst z tego konkretnego obiektu
            zmienna autor_z_bazy = pobierz_pole(wiadomosc, "autor");
            zmienna tekst_z_bazy = pobierz_pole(wiadomosc, "tekst");
            
            // 5. Budujemy piękny dymek czatu!
